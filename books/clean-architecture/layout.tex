\newcommand{\cc@chapterhead}[2]{%
  \vspace*{\cc@pretop}%
  {\parindent\z@\raggedleft
    \rule{\textwidth}{0.75pt}\par
    \addvspace{\cc@afterruleA}%
    {\fontsize{48}{52}\selectfont\bfseries #1\par}%
    \addvspace{\cc@afternum}%
    \rule{\textwidth}{0.75pt}\par
    \addvspace{\cc@afterruleB}%
    {\fontsize{24}{28}\selectfont\bfseries #2\par}%

\def\@makechapterhead#1{%
  \cc@chapterhead{\thechapter}{#1}%
  \setcounter{footnote}{0}%
}
\def\@makeschapterhead#1{%
  \vspace*{\cc@pretop}%
  {\parindent\z@\raggedleft
    \rule{\textwidth}{0.75pt}\par

    {\fontsize{36}{40}\selectfont\bfseries #1\par}%

  \par\nobreak

}

% Unnumbered front/back matter chapter.  \chapter* starts a fresh page, uses
% the 36 pt heading above, and we add the TOC line ourselves (tocloft's
% chapter style is bold 14 pt, meant for numbered chapters).
\newcommand{\ccfrontchapter}[1]{%
  \chapter*{#1}%
  \markboth{#1}{#1}%
  \addtocontents{toc}{\protect\ccfronttocline{#1}{\thepage}}%
}

\newcommand{\ccfronttocline}[2]{%
  \par\noindent
  #1\nobreak\leaders\hbox{\hbox to 4.5pt{\hfil.\hfil}}\hfill\nobreak
  #2\par\vspace{7pt}%
}
