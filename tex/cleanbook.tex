%% cleanbook.tex -- driver for the Vietnamese edition of Clean Code.
%%
%% Build with  ./build.sh  (which runs build/md2tex.py and then xelatex
%% twice).  The chapter bodies are generated fragments under build/gen/.
%%
%% Geometry, headings, code boxes, TOC and running heads are matched to the
%% Pearson 2009 interior (517.608 x 666.36 pt, 360 pt measure, 10/12 Times).
\documentclass[10pt,twoside,openright]{book}

% ---------------------------------------------------------------------------
% Page geometry: trim 517.608 x 666.36 pt, 360 pt measure, symmetric 78.8 pt
% margins, body top 72.4 pt, text height 511 pt, running head at 36.4 pt.
% ---------------------------------------------------------------------------
% PDF measurements are in big points (1bp = 1/72in); TeX's "pt" is 1/72.27in,
% so using pt here would shrink the trim by 0.37%.
\usepackage[
  paperwidth=517.608bp, paperheight=666.36bp,
  textwidth=360bp, textheight=511bp,
  left=78.8bp, right=78.8bp, top=72.4bp, bottom=82.4bp,
  headheight=12bp, headsep=24bp, footskip=24bp,
  twoside=true, columnsep=18bp
]{geometry}

% ---------------------------------------------------------------------------
% Fonts: Tinos (metric clone of Times New Roman), Liberation Sans/Mono.
% ---------------------------------------------------------------------------
\usepackage{fontspec}
\setmainfont{Liberation Serif}[Ligatures=TeX]
\setsansfont{Liberation Sans}[Ligatures=TeX]
\setmonofont{Liberation Mono}[Ligatures=TeX]
% Correct copy/paste and text extraction of Vietnamese diacritics under XeTeX
% (see handoff.md section 6.3).
\XeTeXgenerateactualtext=1

\usepackage{xcolor}
\definecolor{ccink}{HTML}{231F20}
\usepackage{graphicx}
\usepackage{fvextra}

% the printed original has no such mark, and it pollutes the text layer.
\fvset{breakanywheresymbolpre={},breakanywheresymbolpost={},
  breakbeforesymbolpre={},breakbeforesymbolpost={},
  breakaftersymbolpre={},breakaftersymbolpost={}}
\usepackage{tabularx}

% Vertical calibration measured against the original PDF (rules at
% y=90.4pt and 155.4pt from the trim top, chapter number ink at 106.4pt,
% chapter title ink at 204.7pt).
\def\cc@pretop{-2.6pt}
\def\cc@afterruleA{-3.6pt}
\def\cc@afternum{4.6pt}
\def\cc@afterruleB{37.4pt}
\def\cc@fr@afterA{0.3pt}
\def\cc@fr@afterB{12.7pt}
\def\cc@fr@bodygap{35.6pt}

\newcommand{\cc@chapterhead}[2]{%
  \vspace*{\cc@pretop}%
  {\parindent\z@\raggedleft
    \rule{\textwidth}{0.75pt}\par
    \addvspace{\cc@afterruleA}%
    {\fontsize{48}{52}\selectfont\bfseries #1\par}%
    \addvspace{\cc@afternum}%
    \rule{\textwidth}{0.75pt}\par
    \addvspace{\cc@afterruleB}%
    {\fontsize{24}{28}\selectfont\bfseries #2\par}%
  }%
  \par\nobreak
}

\def\@makechapterhead#1{%
  \cc@chapterhead{\thechapter}{#1}%
  \setcounter{footnote}{0}%
}
\def\@makeschapterhead#1{%
  \vspace*{\cc@pretop}%
  {\parindent\z@\raggedleft
    \rule{\textwidth}{0.75pt}\par
    \addvspace{\cc@fr@afterA}%
    {\fontsize{36}{40}\selectfont\bfseries #1\par}%
    \addvspace{\cc@fr@afterB}%
    \rule{\textwidth}{0.75pt}\par
    \addvspace{\cc@fr@bodygap}%
  }%
  \par\nobreak
  \setcounter{footnote}{0}%
}

% Numbered chapter: \ccchapter{title}.  The source may skip chapters (while
% translation is in progress) without renumbering the rest.
\newcommand{\ccchapter}[1]{\chapter{#1}}

% Unnumbered front/back matter chapter.  \chapter* starts a fresh page, uses
% the 36 pt heading above, and we add the TOC line ourselves (tocloft's
% chapter style is bold 14 pt, meant for numbered chapters).
\newcommand{\ccfrontchapter}[1]{%
  \chapter*{#1}%
  \markboth{#1}{#1}%
  \addtocontents{toc}{\protect\ccfronttocline{#1}{\thepage}}%
}

\newcommand{\ccfronttocline}[2]{%
  \par\noindent
  #1\nobreak\leaders\hbox{\hbox to 4.5pt{\hfil.\hfil}}\hfill\nobreak
  #2\par\vspace{7pt}%
}

% Appendices A/B/C: label instead of a number; no opener illustration.
\newcommand{\ccappendixsetup}[1]{%
  \cc@appendixtrue
  \setcounter{chapter}{\numexpr#1-1\relax}%
  \renewcommand{\thechapter}{\Alph{chapter}}%
}
\newcommand{\ccappendix}[1]{%
  \cleardoublepage
  \refstepcounter{chapter}%
  \thispagestyle{plain}%
  \addcontentsline{toc}{chapter}{Phụ lục \thechapter: #1}%
  \markboth{Phụ lục \thechapter: #1}{}%
  \cc@chapterhead{Phụ lục \thechapter}{#1}%
  \setcounter{footnote}{0}%
}
\makeatother

% Byline under the chapter/appendix title ("Tác giả: ...", "với ...").
\newcommand{\ccauthor}[1]{\par{\raggedleft\fontsize{10}{12}\selectfont #1\par}}

% Chapter-opening illustration, width measured from the original PDF.
\newcommand{\ccopenerimage}[2]{%
  \par\addvspace{1pt}%
  {\centering\includegraphics[width=#2]{#1}\par}%
  \addvspace{2pt}%
}

% Uncaptioned inline illustration from the original edition (portraits,
% cartoons, cover photos).  #1 = path, #2 = width with unit.
\newcommand{\ccillustration}[2]{%
  \par\addvspace{2pt}%
  {\centering\includegraphics[width=#2,keepaspectratio]{#1}\par}%
  \addvspace{2pt}%
}

% Chapter epigraph: quotation right aligned, italic; attribution on the last
% line, roman.  Both arguments are LaTeX produced by the converter.
\newcommand{\ccepigraph}[2]{%
  \par\addvspace{8pt}%
  {\raggedleft\itshape\fontsize{10}{12}\selectfont #1\par}%
  \if\relax\detokenize{#2}\relax\else
    \vspace{2pt}%
    {\raggedleft\fontsize{10}{12}\selectfont #2\par}%
  \fi
  \addvspace{10pt}%
}

% Chapter 1's opening paragraph: left aligned over the full measure.
\newcommand{\ccepigraphleft}[2]{%
  \par\addvspace{10pt}%
  {\itshape\fontsize{10}{12}\selectfont #1\par}%
  \if\relax\detokenize{#2}\relax\else
    \vspace{2pt}%
    {\raggedleft\fontsize{10}{12}\selectfont #2\par}%
  \fi
  \addvspace{10pt}%
}

% Appendix C cross-reference line: bold label, dot leader, numbers flush
% right (wrapped numbers stay right aligned, as in the original).
\newcommand{\ccrefitem}[2]{%
  \par\noindent\begingroup
  \fontsize{10}{12}\selectfont
  \makebox[2.0em][l]{#1}%
  \nobreak\leaders\hbox{\hbox to 4pt{\hfil.\hfil}}\hfill\nobreak\
  \setbox0=\hbox{#2}%
  \ifdim\wd0>218pt
    \parbox[t]{218pt}{\raggedleft #2}%
  \else
    \mbox{#2}%
  \fi
  \par\endgroup
  \vspace{0.5pt}%
}

% Pull quote / long blockquote inside the body.
\newenvironment{ccblockquote}{%
  \par\addvspace{8pt}%
  \begin{list}{}{%
    \setlength{\leftmargin}{24pt}%
    \setlength{\rightmargin}{12pt}%
    \setlength{\topsep}{0pt}%
    \setlength{\parsep}{4pt}%
    \setlength{\itemsep}{0pt}%
    \setlength{\itemindent}{0pt}%
    \setlength{\listparindent}{1em}%
  }%
  \item\relax\itshape
}{%
  \end{list}%
  \addvspace{8pt}%
}

% Figure floats-less block: centred image + bold "Hình X-Y" caption, then the

}
